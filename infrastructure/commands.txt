# Unified Make & Task Commands (Run All Tests with 3 Commands)

# Run Unit Tests across all services and packages:
make test-unit
# or with Taskfile:
# task test-unit
# or for a specific service / package:
make test-unit SERVICE=user-service
make test-unit SERVICE=payment-service
make test-unit SERVICE=packages/go

# Run E2E Tests across all services:
make test-e2e
# or with Taskfile:
# task test-e2e
# or for a specific service:
make test-e2e SERVICE=payment-service
make test-e2e SERVICE=api-gateway

# Run Test Coverage across all services and packages:
make test-cov
# or with Taskfile:
# task test-cov
# or for a specific service / package:
make test-cov SERVICE=user-service
make test-cov SERVICE=driver-service

# Run All Services in dev mode (Docker Compose)
make dev-up ENV=development
make dev-down ENV=development
make dev-logs
make dev-ps
# Direct compose equivalents:
docker compose --env-file config/env/.env.development -f infrastructure/docker/compose.yml -f infrastructure/docker/compose.integration.yml up --build -d
docker compose --env-file config/env/.env.development -f infrastructure/docker/compose.yml -f infrastructure/docker/compose.integration.yml down
docker compose --env-file config/env/.env.development -f infrastructure/docker/compose.yml ps
docker compose --env-file config/env/.env.development -f infrastructure/docker/compose.yml logs -f <service-name>
docker compose --env-file config/env/.env.development -f infrastructure/docker/compose.yml exec <service-name> sh

# Run Infrastructure Separate (DBs, Redis, Kafka, NATS, etc.)
